cmake_minimum_required(VERSION 3.8)
project(roboquest_description)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(heartbeat src/heartbeat.cpp)
target_compile_features(heartbeat PUBLIC cxx_std_17)
ament_target_dependencies(heartbeat rclcpp std_msgs)
install(TARGETS heartbeat DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/joint_demo.py scripts/tf_probe.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch urdf DESTINATION share/${PROJECT_NAME})
ament_package()
